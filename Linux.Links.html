Beispiellösung der Übungsaufgabe

Kommandos

mkdir ~/bin
touch ~/bin/mein_bash-script && chmod u+x ~/bin/mein_bash-script && ll ~/bin/mein_bash-script
nano ~/bin/mein_bash-script
cat ~/bin/mein_bash-script
stat ~/bin/mein_bash-script

Inhalt von ~/bin/mein_bash-script

#!/bin/bash
# -*- coding: utf-8 -*-
# Übungsskript von hardi.kirsten@daa.de
#
Folder="~/Aufgabenverzeichnis/$(date +%F)"
mkdir -p "${Folder}"
#
touch "${Folder}"/besitzer-lesen && chmod a=,u=r "${Folder}"/besitzer-lesen
touch "${Folder}"/besitzer-schreiben && chmod a=,u=w "${Folder}"/besitzer-schreiben
touch "${Folder}"/besitzer-ausführen && chmod a=,u=x "${Folder}"/besitzer-ausführen
#
touch "${Folder}"/gruppe-lesen && chmod a=,g=r "${Folder}"/gruppe-lesen
touch "${Folder}"/gruppe-schreiben && chmod a=,g=w "${Folder}"/gruppe-schreiben
touch "${Folder}"/gruppe-ausführen && chmod a=,g=x "${Folder}"/gruppe-ausführen
#
touch "${Folder}"/andere-lesen && chmod a=,o=r "${Folder}"/andere-lesen
touch "${Folder}"/andere-schreiben && chmod a=,o=w "${Folder}"/andere-schreiben
touch "${Folder}"/andere-ausführen && chmod a=,o=x "${Folder}"/andere-ausführen
#
touch "${Folder}"/besitzer-alles && chmod a=,u=rwx "${Folder}"/besitzer-alles
touch "${Folder}"/gruppe-alles && chmod a=,g=rwx "${Folder}"/gruppe-alles
touch "${Folder}"/andere-alles && chmod a=,o=rwx "${Folder}"/andere-alles
### eof ###

SSH-Daemon installieren und aktivieren

Kommandos für ein Skript "install_sshd.sh"

#!/bin/bash
# Skript zur Installation und Aktivierung des SSH-Daemons
# Skript muss ausgeführt werden
#
apt install -y openssh-server # openSSHD installieren
systemctl enable ssh # Dienst SSHD in Startupkonfiguration eintragen
systemctl start ssh # Dienst SSHD starten
ufw allow ssh # Port 22 in der Firewall freischalten
